// middleware.ts

import { NextResponse, type NextRequest } from 'next/server'
import { createServerClient } from '@supabase/ssr'

// Cesty, které vyžadují přihlášení (role 'user' nebo 'admin')
const PROTECTED_PATHS_FOR_USER = [
'/user-portal',
'/documents',
'/chat',
'/tables'
];

// Cesty, které vyžadují roli 'admin'
const PROTECTED_PATHS_FOR_ADMIN = [
'/admin',
];

export async function middleware(request: NextRequest) {

    // Vytvoření server klienta pro přístup k relaci (session) přes cookies
    const supabase = createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
            cookies: {
                get: (name) => request.cookies.get(name)?.value,
                set: (name, value, options) => {
                    // Middleware vrací upravenou odpověď, která nastavuje cookie
                    request.cookies.set({ name, value, ...options })
                },
                remove: (name, options) => {
                    request.cookies.set({ name, value: '', ...options })
                },
            },
        }
    )

    // Získání relace z cookies
    const { data: { session } } = await supabase.auth.getSession();
    const pathname = request.nextUrl.pathname;

    // --- LOGIKA PRO PŘESMĚROVÁNÍ ---

    // 1. Ochrana cest pro "user" a "admin"
    const isProtectedUserPath = PROTECTED_PATHS_FOR_USER.some(path => pathname.startsWith(path));

    if (isProtectedUserPath && !session) {
        // Uživatel není přihlášen a snaží se dostat na chráněnou cestu
        const redirectUrl = request.nextUrl.clone();
        redirectUrl.pathname = '/prihlaseni';
        // Přidáme původní cestu do URL, aby se po přihlášení mohl vrátit
        redirectUrl.searchParams.set(`redirectTo`, pathname);
        return NextResponse.redirect(redirectUrl);
    }

    // Zde by normálně proběhla kontrola role:
    // 2. Ochrana cest pro "admin" (Vyžaduje načtení role z DB)

    if (session) {
        // Zde byste volali Supabase DB pro získání role uživatele:
         const { data: profile } = await supabase.from('profiles').select('role').eq('id', session.user.id).single();
         const userRole = profile?.role; // 'user' nebo 'admin'

        const isAdminPath = PROTECTED_PATHS_FOR_ADMIN.some(path => pathname.startsWith(path));

        if (isAdminPath && userRole !== 'admin') {
            // Uživatel je přihlášen, ale nemá roli 'admin' pro cestu /admin
            const redirectUrl = request.nextUrl.clone();
            redirectUrl.pathname = '/'; // Přesměrování na úvod
            return NextResponse.redirect(redirectUrl);
        }
    }


    // 3. Přihlášený uživatel na stránce /prihlaseni
    if (session && pathname === '/prihlaseni') {
        // Přihlášeného uživatele posíláme pryč z přihlašovací stránky
        const redirectUrl = request.nextUrl.clone();
        // Vracíme ho tam, odkud přišel, nebo na domovskou stránku portálu
        redirectUrl.pathname = request.nextUrl.searchParams.get('redirectTo') || '/user-portal/dokumenty';
        redirectUrl.searchParams.delete('redirectTo');
        return NextResponse.redirect(redirectUrl);
    }

    // Pokračování na požadovanou cestu
    return NextResponse.next();

}

// src/app/admin/contact/page.tsx
'use client';

import React, { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/lib/supabase/client';
import { Mail, Trash2, Eye, EyeOff, Loader2, CheckCircle, XCircle } from 'lucide-react';

// Typ pro jednu zprávu z tabulky contact_submissions
interface Submission {
id: string;
name: string;
email: string;
message: string;
visible: boolean;
created_at: string;
}

const ContactManager: React.FC = () => {
const [submissions, setSubmissions] = useState<Submission[]>([]);
const [isLoading, setIsLoading] = useState(true);
const [error, setError] = useState<string | null>(null);

    const fetchSubmissions = useCallback(async () => {
        setIsLoading(true);
        setError(null);

        // Načteme všechny zprávy, nejnovější první
        const { data, error } = await supabase
            .from('contact_form')
            .select('*')
            .order('created_at', { ascending: false });

        if (error) {
            console.error('Chyba při načítání zpráv:', error);
            setError('Nepodařilo se načíst zprávy. Zkontrolujte RLS politiky (SELECT pro admina).');
            setSubmissions([]);
        } else {
            setSubmissions(data || []);
        }
        setIsLoading(false);
    }, []);

    // Načtení dat při prvním renderu
    useEffect(() => {
        fetchSubmissions();
    }, [fetchSubmissions]);

    // --- MANIPULAČNÍ FUNKCE ---

    const toggleReadStatus = async (id: string, currentStatus: boolean) => {
        const newStatus = !currentStatus;

        // Optimistická aktualizace UI
        setSubmissions(prev =>
            prev.map(sub => sub.id === id ? { ...sub, visible: newStatus } : sub)
        );

        const { error } = await supabase
            .from('contact_form')
            .update({ visible: newStatus })
            .eq('id', id);

        if (error) {
            console.error('Chyba při změně statusu:', error);
            // V případě chyby vrátit stav zpět
            setSubmissions(prev =>
                prev.map(sub => sub.id === id ? { ...sub, visible: currentStatus } : sub)
            );
            alert('Chyba při aktualizaci statusu: ' + error.message);
        }
    };

    const deleteSubmission = async (id: string) => {
        if (!confirm('Opravdu chcete tuto kontaktní zprávu smazat?')) {
            return;
        }

        const originalSubmissions = submissions;

        // Optimistická aktualizace UI
        setSubmissions(prev => prev.filter(sub => sub.id !== id));

        const { error } = await supabase
            .from('contact_form')
            .delete()
            .eq('id', id);

        if (error) {
            console.error('Chyba při mazání zprávy:', error);
            setSubmissions(originalSubmissions); // Vrátit zprávy v případě chyby
            alert('Chyba při mazání zprávy: ' + error.message);
        }
    };

    if (isLoading) {
        return <div className="p-8 text-center"><Loader2 className="w-6 h-6 animate-spin mx-auto text-blue-500" /> Načítám zprávy...</div>;
    }

    if (error) {
        return <div className="p-8 text-red-600 border border-red-300 bg-red-50 rounded-lg">Chyba: {error}</div>;
    }

    return (
        <div className="bg-white p-6 rounded-lg shadow-md">
            <h2 className="max-w-7xl mx-auto text-2xl font-bold text-gray-800 mb-6 flex items-center dark:text-gray-100 border-b pb-2">
                <Mail className="w-6 h-6 mr-2 text-blue-600" />
                Kontaktní Zprávy ({submissions.length})
            </h2>

            {submissions.length === 0 ? (
                <div className="text-center py-10 text-gray-500">
                    <CheckCircle className="w-10 h-10 mx-auto text-green-500 mb-3" />
                    Všechny zprávy jsou vyřízeny. Vypadá to tu čistě!
                </div>
            ) : (
                <div className="space-y-4">
                    {submissions.map((sub) => (
                        <div
                            key={sub.id}
                            className={`p-4 border rounded-lg transition duration-200 ${
                                sub.visible ? 'bg-gray-100 border-gray-200 opacity-80' : 'bg-yellow-50 border-yellow-300 font-medium'
                            }`}
                        >
                            <div className="flex justify-between items-start mb-2">
                                <div>
                                    <p className="text-lg text-gray-900">
                                        Od: <span className="font-bold">{sub.name}</span>
                                    </p>
                                    <p className="text-sm text-gray-600">
                                        <a href={`mailto:${sub.email}`} className="hover:underline">{sub.email}</a>
                                        {' | '}
                                        <span title="Čas odeslání">
                                            {new Date(sub.created_at).toLocaleString('cs-CZ')}
                                        </span>
                                    </p>
                                </div>
                                <div className="flex space-x-2 flex-shrink-0">
                                    <button
                                        onClick={() => toggleReadStatus(sub.id, sub.visible)}
                                        className={`p-2 rounded-full transition duration-150 ${
                                            sub.visible ? 'text-gray-500 hover:bg-gray-200' : 'text-blue-600 hover:bg-blue-100'
                                        }`}
                                        title={sub.visible ? 'Označit jako nepřečtené' : 'Označit jako přečtené'}
                                    >
                                        {sub.visible ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                                    </button>
                                    <button
                                        onClick={() => deleteSubmission(sub.id)}
                                        className="p-2 rounded-full text-red-500 hover:bg-red-100 transition duration-150"
                                        title="Smazat zprávu"
                                    >
                                        <Trash2 className="w-5 h-5" />
                                    </button>
                                </div>
                            </div>
                            <div className={`mt-3 pt-3 border-t ${sub.visible ? 'border-gray-300' : 'border-yellow-400'}`}>
                                <p className="whitespace-pre-wrap text-gray-700">{sub.message}</p>
                            </div>
                        </div>
                    ))}
                </div>
            )}
        </div>
    );

};

export default ContactManager;

interface Announcement {

interface AnnouncementModalProps {
isOpen: boolean;
onClose: () => void;
initialData: Announcement | null; // null pro přidání, data pro editaci
onSaveSuccess: () => void;
}

const AnnouncementModal: React.FC<AnnouncementModalProps> = ({ isOpen, onClose, initialData, onSaveSuccess }) => {
const [formData, setFormData] = useState<Omit<Announcement, 'id' | 'created_at'>>({
title: '',
content: '',
is_active: true,
priority: 1,
target_role: 'all',
});
const [isSaving, setIsSaving] = useState(false);
const [error, setError] = useState<string | null>(null);

    // Načtení dat při otevření modalu pro editaci
    useEffect(() => {
        if (initialData) {
            setFormData({
                title: initialData.title,
                content: initialData.content,
                is_active: initialData.is_active,
                priority: initialData.priority,
                target_role: initialData.target_role,
            });
        } else {
            // Reset pro přidání nového
            setFormData({ title: '', content: '', is_active: true, priority: 1, target_role: 'all' });
        }
        setError(null);
    }, [initialData]);

    if (!isOpen) return null;

    const handleSave = async () => {
        if (!formData.title || !formData.content) {
            setError('Titulek a obsah nesmí být prázdné.');
            return;
        }

        setIsSaving(true);
        setError(null);

        try {
            if (initialData) {
                // EDITACE EXISTUJÍCÍHO
                const { error: updateError } = await supabase
                    .from('portal_announcements')
                    .update(formData)
                    .eq('id', initialData.id);

                if (updateError) throw updateError;
            } else {
                // PŘIDÁNÍ NOVÉHO
                const { error: insertError } = await supabase
                    .from('portal_announcements')
                    .insert(formData);

                if (insertError) throw insertError;
            }

            onSaveSuccess();
            onClose();

        } catch (err: any) {
            console.error('Chyba při ukládání oznámení:', err);
            setError(`Chyba: ${err.message || 'Nepodařilo se uložit oznámení.'}`);
        } finally {
            setIsSaving(false);
        }
    };

    return (
        <div
            className="fixed inset-0 z-50 flex items-center justify-center bg-black bg-opacity-70 backdrop-blur-sm p-4"
            onClick={onClose}
        >
            <div

                    {/* Titulek */}

                        <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Titulek</label>

                            disabled={isSaving}
                        />
                    </div>

                    {/* Obsah */}
                    <div>
                        <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Obsah (podporuje jednoduché formátování)</label>

                            value={formData.content}
                            onChange={(e) => setFormData(prev => ({ ...prev, content: e.target.value }))}
                            rows={4}
                            className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"

                                onChange={(e) => setFormData(prev => ({ ...prev, priority: parseInt(e.target.value) || 1 }))}
                                min="1"
                                className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"
                                disabled={isSaving}

            setError('Přístup odepřen: Musíte být administrátor.');
            setIsLoading(false);
            return;
        }

        const { data, error: fetchError } = await supabase
            .from('portal_announcements')
            .select('*')
            .order('priority', { ascending: true })
            .order('created_at', { ascending: false });

        if (fetchError) {
            console.error('Chyba při načítání oznámení:', fetchError);
            setError('Nepodařilo se načíst oznámení.');
            setAnnouncements([]);
        } else {
            setAnnouncements(data || []);
        }
        setIsLoading(false);

        fetchAnnouncements();

        setEditingData(announcement);
        setIsModalOpen(true);

        setEditingData(null);
        setIsModalOpen(true);

                .from('portal_announcements')
                .delete()

            // Supabase Cascade Delete by se mělo postarat o smazání záznamů z 'user_announcement_reads'

        }));
    };

    // --- FUNKCE PRO NAČTENÍ PROFILŮ ---
    const fetchProfiles = useCallback(async () => {
        setIsLoading(true);
        setError(null);

        const { data: profilesData, error: fetchError } = await supabase
            .from('profiles')
            .select('id, full_name, email, role, status, is_blocked')
            .order('full_name', { ascending: true });

        if (fetchError) {
            console.error('Chyba při načítání profilů:', fetchError);
            setError('Nepodařilo se načíst uživatelské profily.');
        } else {
            setProfiles(formatProfiles(profilesData || []));
        }
        setIsLoading(false);
    }, []);

    // --- FUNKCE PRO AKTUALIZACI PROFILU ---
    const updateProfile = async (id: string, updates: Partial<Profile>) => {
        if (!user) return;

        // Extrahujeme pouze sloupce relevantní pro databázi
        const dbUpdates: Partial<Omit<Profile, 'email'>> = {
            full_name: updates.full_name,
            role: updates.role,
            status: updates.status,
            is_blocked: updates.is_blocked,
        };

        const { error: updateError } = await supabase
            .from('profiles')
            .update(dbUpdates)
            .eq('id', id);

        if (updateError) {
            setError(`Chyba při aktualizaci profilu ${id}: ${updateError.message}`);
            return;
        }

        // Aktualizujeme stav lokálně
        setProfiles(prev =>
            prev.map(p => {
                if (p.id === id) {
                    // Důležité: Upravit status, pokud se mění is_blocked
                    let newStatus = updates.status || p.status;
                    if (updates.is_blocked !== undefined) {
                        newStatus = updates.is_blocked ? 'blocked' : (p.status === 'blocked' ? 'pending' : p.status);
                    }
                    return { ...p, ...updates, status: newStatus };
                }
                return p;
            })
        );
        setError(null);
    };

    useEffect(() => {
        fetchProfiles();
    }, [fetchProfiles]);


    // --- FILTRACE PROFILŮ PODLE AKTIVNÍ ZÁLOŽKY ---
    const filteredProfiles = useMemo(() => {
        if (activeFilter === 'all') return profiles;

        return profiles.filter(p => {
            if (activeFilter === 'blocked') {
                // Filtrujeme, pokud je uživatel blokovaný
                return p.is_blocked;
            }
            if (activeFilter === 'approved') {
                // Filtrujeme, pokud je schválený A NENÍ blokovaný
                return p.status === 'approved' && !p.is_blocked;
            }
            if (activeFilter === 'pending') {
                 // Filtrujeme, pokud čeká na schválení A NENÍ blokovaný
                return p.status === 'pending' && !p.is_blocked;
            }
            return false;
        });
    }, [profiles, activeFilter]);

    // Definice záložek
    const tabs: { name: UserFilter, label: string, count: number }[] = useMemo(() => [
        { name: 'pending', label: 'Neschválení', count: profiles.filter(p => p.status === 'pending' && !p.is_blocked).length },
        { name: 'approved', label: 'Schválení', count: profiles.filter(p => p.status === 'approved' && !p.is_blocked).length },
        { name: 'blocked', label: 'Blokovaní', count: profiles.filter(p => p.is_blocked).length },
        { name: 'all', label: 'Všichni', count: profiles.length },
    ], [profiles]);


    // --- RENDERING OVLÁDÁNÍ STAVU A CHYB ---
    if (isLoading) {
        return (
            <div className="flex justify-center items-center h-64">
                <GridLoader color="#0066ff" />
            </div>
        );
    }

    if (error) {
        return <div className="p-4 bg-red-100 text-red-700 rounded dark:bg-red-900 dark:text-red-300">Chyba: {error}</div>;
    }

    const getStatusStyle = (profile: Profile) => {
        if (profile.is_blocked) return 'bg-red-500 text-white';
        if (profile.status === 'approved') return 'bg-green-100 text-green-800';
        if (profile.status === 'pending') return 'bg-yellow-100 text-yellow-800';
        return 'bg-gray-100 text-gray-800';
    };


    // --- RENDERING KOMPONENTY ---
    return (
        <div className="bg-white dark:bg-zinc-800 p-6 rounded-lg shadow-xl">
            <h2 className="text-2xl font-bold mb-6 text-gray-800 dark:text-gray-100 border-b pb-2">Správa Uživatelů 👤</h2>

            {/* Navigace záložkami (Filtry) */}
            <nav className="mb-6 border-b border-gray-200 dark:border-zinc-700">
                <div className="flex space-x-4 overflow-x-auto">
                    {tabs.map(tab => (
                        <button
                            key={tab.name}
                            onClick={() => setActiveFilter(tab.name)}
                            className={`
                                py-2 px-4 text-sm font-semibold transition-colors duration-150 relative
                                ${activeFilter === tab.name
                                    ? 'text-blue-600 border-b-2 border-blue-600 dark:text-blue-400 dark:border-blue-400'
                                    : 'text-gray-500 dark:text-zinc-400 hover:text-gray-700 dark:hover:text-zinc-200'
                                }
                            `}
                        >
                            {tab.label} ({tab.count})
                        </button>
                    ))}
                </div>
            </nav>

            {/* Obsah tabulky */}
            {filteredProfiles.length === 0 ? (
                <p className="p-4 text-gray-500 dark:text-zinc-400">V této sekci nejsou žádní uživatelé k zobrazení.</p>
            ) : (
                <div className="overflow-x-auto">
                    <table className="min-w-full divide-y divide-gray-200 dark:divide-zinc-700">
                        <thead className="bg-gray-50 dark:bg-zinc-700">
                            <tr>
                                <th className="px-6 py-3 text-left text-xs font-medium text-gray-500 dark:text-zinc-300 uppercase tracking-wider">Uživatel</th>
                                <th className="px-6 py-3 text-center text-xs font-medium text-gray-500 dark:text-zinc-300 uppercase tracking-wider">Role</th>
                                <th className="px-6 py-3 text-center text-xs font-medium text-gray-500 dark:text-zinc-300 uppercase tracking-wider">Stav</th>
                                <th className="px-6 py-3 text-center text-xs font-medium text-gray-500 dark:text-zinc-300 uppercase tracking-wider">Akce</th>
                            </tr>
                        </thead>
                        <tbody className="bg-white dark:bg-zinc-800 divide-y divide-gray-200 dark:divide-zinc-700">
                            {filteredProfiles.map((profile) => (
                                <tr
                                    key={profile.id}
                                    className={`
                                        transition-colors hover:bg-gray-50 dark:hover:bg-zinc-700/50
                                        ${profile.is_blocked ? 'opacity-70' : ''}
                                    `}
                                >
                                    {/* Jméno a Email */}
                                    <td className="px-6 py-4 whitespace-nowrap">
                                        <div className="text-sm font-medium text-gray-900 dark:text-zinc-100">{profile.full_name || 'N/A'}</div>
                                        <div className="text-sm text-gray-500 dark:text-zinc-400">{profile.email}</div>
                                    </td>

                                    {/* Role (Select) */}
                                    <td className="px-6 py-4 whitespace-nowrap text-center">
                                        <select
                                            value={profile.role}
                                            onChange={(e) => updateProfile(profile.id, { role: e.target.value as 'admin' | 'user' })}
                                            className={`p-1 rounded border text-sm font-medium dark:bg-zinc-700 dark:border-zinc-600
                                            ${profile.role === 'admin' ? 'bg-red-100 text-red-700 dark:text-red-300' : 'bg-green-100 text-green-700 dark:text-green-300'}`}
                                            disabled={profile.is_blocked}
                                        >
                                            <option value="user">USER</option>
                                            <option value="admin">ADMIN</option>
                                        </select>
                                    </td>

                                    {/* Stav (Status Label) */}
                                    <td className="px-6 py-4 whitespace-nowrap text-center">
                                        <span className={`px-3 py-1 rounded-full text-xs font-semibold ${getStatusStyle(profile)}`}>
                                            {profile.status.toUpperCase()}
                                        </span>
                                    </td>

                                    {/* Akce */}
                                    <td className="px-6 py-4 whitespace-nowrap text-center text-sm font-medium space-x-2">

                                        {/* Tlačítko pro Schválení/Neschválení */}
                                        {profile.status !== 'approved' && !profile.is_blocked && (
                                            <button
                                                onClick={() => updateProfile(profile.id, { status: 'approved' })}
                                                className="p-2 rounded-full text-green-600 hover:bg-green-100 dark:hover:bg-green-900/50 transition"
                                                title="Schválit uživatele"
                                            >
                                                <UserCheck size={20} />
                                            </button>
                                        )}
                                        {profile.status === 'approved' && !profile.is_blocked && (
                                            <button
                                                onClick={() => updateProfile(profile.id, { status: 'pending' })}
                                                className="p-2 rounded-full text-yellow-600 hover:bg-yellow-100 dark:hover:bg-yellow-900/50 transition"
                                                title="Přesunout do čekajících"
                                            >
                                                <X size={20} />
                                            </button>
                                        )}

                                        {/* Tlačítko pro Blokování/Odblokování */}
                                        <button
                                            onClick={() => updateProfile(profile.id, { is_blocked: !profile.is_blocked, status: !profile.is_blocked ? 'blocked' : 'pending' })}
                                            className={`p-2 rounded-full transition ${
                                                profile.is_blocked
                                                ? 'text-white bg-red-600 hover:bg-red-700'
                                                : 'text-red-600 hover:bg-red-100 dark:hover:bg-red-900/50'
                                            }`}

                                        >
                                            {profile.is_blocked ? <Check size={20} /> : <UserX size={20} />}
                                        </button>
                                    </td>
                                </tr>
                            ))}
                        </tbody>
                    </table>
                </div>
            )}
        </div>

admin_only: boolean;
type: 'smernice' | 'manualy' | 'jine';

// ----------------------------------------------------------------------
// --- PLACEHOLDER PRO MODAL PRO ÚPRAVU (musí být vytvořen jako samostatná komponenta) ---
// ----------------------------------------------------------------------

interface EditDocumentModalProps {
isOpen: boolean;
onClose: () => void;
document: Document | null;
onSave: (updatedDoc: Partial<Document>) => void;
}

// 💡 Zde byste normálně importovali skutečnou komponentu modalu.
// V tomto příkladu necháme jednoduchý placeholder pro správnou funkci.
const EditDocumentModal: React.FC<EditDocumentModalProps> = ({ isOpen, onClose, document, onSave }) => {
// Lokální stav pro data formuláře, inicializovaný z `document`
const [formData, setFormData] = useState<Partial<Document>>({});
const [isSaving, setIsSaving] = useState(false);

    useEffect(() => {
        // Reset formuláře při otevření modalu nebo změně dokumentu
        if (document) {
            setFormData({
                title: document.title,
                description: document.description,
                type: document.type,
                admin_only: document.admin_only,
            });
        }
    }, [document]);

    if (!isOpen || !document) return null;

    const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) => {
        const { name, value, type } = e.target;

        // Speciální ošetření pro checkbox
        const inputValue = type === 'checkbox'
            ? (e.target as HTMLInputElement).checked
            : value;

        setFormData(prev => ({
            ...prev,
            [name]: inputValue,
        }));
    };

    const handleSubmit = async (e: FormEvent) => {
        e.preventDefault();
        setIsSaving(true);

        // Filtrování pouze změněných hodnot
        const changes: Partial<Document> = {};
        let isChanged = false;

        (['title', 'description', 'type', 'admin_only'] as Array<keyof Document>).forEach(key => {
            // TypeScript bezpečné porovnání
            const currentValue = document[key];
            const newValue = formData[key];

            // Kontrola změn (zejména pro null/string a boolean)
            if (newValue !== undefined && newValue !== currentValue) {
                changes[key] = newValue as any;
                isChanged = true;
            }
        });

        try {
            if (isChanged) {
                await onSave(changes); // Volání funkce pro uložení v DocumentsPage
            } else {
                alert('Nebyly zjištěny žádné změny k uložení.');
                onClose();
            }
        } catch (err) {
            console.error(err);
            alert('Chyba při ukládání dokumentu.');
        } finally {
            setIsSaving(false);
        }
    };

    // Tailwind pro dark mode
    const inputClasses = "block w-full border border-gray-300 dark:border-zinc-600 rounded-lg shadow-sm p-2 focus:border-blue-500 focus:ring-blue-500 dark:bg-zinc-700 dark:text-zinc-100 transition";
    const labelClasses = "block text-sm font-medium text-gray-700 dark:text-zinc-300 mb-1";

    return (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
            <div className="bg-white dark:bg-zinc-800 p-6 rounded-lg shadow-2xl max-w-lg w-full relative">
                <button
                    onClick={onClose}
                    className="absolute top-3 right-3 text-gray-500 dark:text-zinc-400 hover:text-gray-800 dark:hover:text-zinc-100"
                    disabled={isSaving}
                >
                    <X className="w-6 h-6" />
                </button>

                <h2 className="text-xl font-bold mb-4 text-gray-800 dark:text-gray-100">Upravit dokument</h2>

                <form onSubmit={handleSubmit}>
                    {/* Název */}
                    <div className="mb-4">
                        <label htmlFor="title" className={labelClasses}>Název</label>
                        <input
                            type="text"
                            id="title"
                            name="title"
                            value={formData.title || ''}
                            onChange={handleChange}
                            className={inputClasses}
                            required
                        />
                    </div>

                    {/* Popis */}
                    <div className="mb-4">
                        <label htmlFor="description" className={labelClasses}>Popis (volitelné)</label>
                        <textarea
                            id="description"
                            name="description"
                            value={formData.description || ''}
                            onChange={handleChange}
                            className={`${inputClasses} h-20`}
                        />
                    </div>

                    {/* Typ */}
                    <div className="mb-4">
                        <label htmlFor="type" className={labelClasses}>Typ Dokumentu</label>
                        <select
                            id="type"
                            name="type"
                            value={formData.type || 'jine'}
                            onChange={handleChange}
                            className={inputClasses}
                            required
                        >
                            <option value="smernice">Směrnice</option>
                            <option value="manualy">Manuály</option>
                            <option value="jine">Jiné</option>
                        </select>
                    </div>

                    {/* Pouze pro Adminy */}
                    <div className="mb-6 flex items-center">
                        <input
                            type="checkbox"
                            id="admin_only"
                            name="admin_only"
                            checked={!!formData.admin_only} // !! pro zajištění boolean hodnoty
                            onChange={handleChange}
                            className="w-4 h-4 text-blue-600 bg-gray-100 border-gray-300 rounded focus:ring-blue-500 dark:focus:ring-blue-600 dark:ring-offset-gray-800 dark:bg-gray-700 dark:border-gray-600"
                        />

                            Dokument je pouze pro Adminy
                        </label>
                    </div>

                    {/* Tlačítka */}

                        <button
                            type="button"
                            onClick={onClose}

                  <div className="flex items-start space-x-3 sm:space-x-4 w-full sm:w-auto">
                    {getTypeIcon(doc.type as DocumentType)}
                    <div className="flex-1 min-w-0">
                      <h3 className="text-base sm:text-lg font-bold text-gray-800 dark:text-zinc-50 truncate">{doc.title}</h3>
                      <p className="text-xs sm:text-sm text-gray-500 dark:text-zinc-400 mt-1 line-clamp-2">{doc.description}</p>
                      <p className="text-xs text-gray-400 dark:text-zinc-500 mt-1">
                        Typ: {getTypeName(doc.type as DocumentType)} | Nahráno: {new Date(doc.created_at).toLocaleDateString()}
                          {doc.admin_only && <span className="text-red-500 dark:text-red-400 ml-2 font-semibold">(Admin Only)</span>}
                      </p>
                    </div>
                  </div>

                  {/* Akční Tlačítka (Download + Admin Actions) */}
                  <div className="mt-4 sm:mt-0 sm:ml-6 shrink-0 flex gap-2 w-full sm:w-auto justify-end sm:justify-start">
                        {/* 💡 Admin Tlačítka (Úprava a Smazání) */}
                        {isAdmin && (
                            <>
                                <button
                                    onClick={() => handleEdit(doc)} // Otevře modal s daty dokumentu
                                    className="p-2 bg-yellow-500 text-white rounded-lg hover:bg-yellow-600 transition"
                                    title="Upravit dokument"
                                >
                                    <Edit className="w-4 h-4" />
                                </button>
                                <button
                                    onClick={() => handleDelete(doc.id, doc.file_path)} // Smaže dokument
                                    className="p-2 bg-red-600 text-white rounded-lg hover:bg-red-700 transition"
                                    title="Smazat dokument"
                                >
                                    <Trash2 className="w-4 h-4" />
                                </button>
                            </>
                        )}

                        {/* Tlačítko pro stažení */}
                        <Link
                          href={getDownloadUrl(doc.file_path)}
                          target="_blank"
                          rel="noopener noreferrer"
                          className="flex items-center justify-center space-x-2 px-4 py-2 bg-green-600 text-white rounded-lg font-semibold hover:bg-green-700 transition w-full sm:w-auto"
                        >
                          <Download className="w-4 h-4" />
                          <span>Stáhnout</span>
                        </Link>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>

        {/* 💡 Vykreslení Modalu pro úpravu */}
        <EditDocumentModal
            isOpen={isModalOpen}
            onClose={() => setIsModalOpen(false)}
            document={editingDocument}
            onSave={handleSaveEdit}
        />
      </div>
    );

};

// === 3. KOMPONENTA MODALU PRO EDITACI (NOVÁ PLNOU IMPLEMENTACÍ) ===
// ======================================================================
interface EditTimetableModalProps {
isOpen: boolean;
onClose: () => void;
initialData: GroupedTimetable | null;
onSaveSuccess: () => void;
}

const EditTimetableModal: React.FC<EditTimetableModalProps> = ({ isOpen, onClose, initialData, onSaveSuccess }) => {

    const [formData, setFormData] = useState({
        line_number: '',
        course_number: '',
        transport_type: 'autobus' as 'autobus' | 'tramvaj',
        day_type: 'pracovni' as 'pracovni' | 'vikend',
    });

    const [fileRanni, setFileRanni] = useState<File | null>(null);
    const [fileOdpoledni, setFileOdpoledni] = useState<File | null>(null);
    const [isSaving, setIsSaving] = useState(false);
    const [statusMessage, setStatusMessage] = useState<string | null>(null);

    useEffect(() => {
        if (initialData) {
            setFormData({
                line_number: initialData.line_number,
                course_number: initialData.course_number,
                transport_type: initialData.transport_type,
                day_type: initialData.day_type,
            });
            setFileRanni(null);
            setFileOdpoledni(null);
            setStatusMessage(null);
        }
    }, [initialData]);

    if (!isOpen || !initialData) return null;

    // Funkce pro nahrání souboru a získání veřejné URL
    const uploadFile = async (file: File, part: 'ranni' | 'odpoledni', oldRecord: JizdniRad | null) => {
        const fileExt = file.name.split('.').pop();
        // Název souboru je klíčový pro přepsání v Supabase Storage!
        const fileName = `${formData.line_number}_${formData.course_number}_${formData.day_type}_${part}.${fileExt}`;
        const filePath = `${fileName}`;

        // 1. Nahrání nového souboru do Storage (přepíše stávající soubor se stejným jménem)
        const { data: uploadData, error: uploadError } = await supabase.storage
            .from('jizdni-rady') // Jméno vašeho bucketu
            .upload(filePath, file, {
                cacheControl: '3600',
                upsert: true, // Klíčové: pokud soubor existuje, nahradí ho
            });

        if (uploadError) {
            throw new Error(`Chyba při nahrávání souboru ${part}: ${uploadError.message}`);
        }

        // 2. Získání veřejné URL
        const { data: publicUrlData } = supabase.storage
            .from('jizdni-rady')
            .getPublicUrl(filePath);

        // 3. Aktualizace/Vložení záznamu do tabulky jizdni_rady
        const newRecordData = {
            line_number: formData.line_number,
            course_number: formData.course_number,
            transport_type: formData.transport_type,
            day_type: formData.day_type,
            part_of_day: part,
            image_url: publicUrlData.publicUrl,
        };

        if (oldRecord) {
            // AKTUALIZACE STÁVAJÍCÍHO ZÁZNAMU
            const { error: updateError } = await supabase
                .from('jizdni_rady')
                .update(newRecordData)
                .eq('id', oldRecord.id);

            if (updateError) throw new Error(`Chyba při aktualizaci DB záznamu ${part}: ${updateError.message}`);
        } else {
            // VLOŽENÍ NOVÉHO ZÁZNAMU (pokud předtím neexistoval)
            const { error: insertError } = await supabase
                .from('jizdni_rady')
                .insert(newRecordData);

            if (insertError) throw new Error(`Chyba při vkládání DB záznamu ${part}: ${insertError.message}`);
        }
    };

    // Hlavní funkce pro uložení změn
    const handleSave = async () => {
        setIsSaving(true);
        setStatusMessage(null);

        try {
            // Kontrola platnosti dat (přidáno pro robustnost)
            if (!formData.line_number || !formData.course_number) {
                 throw new Error('Číslo Linky a Kurzu nesmí být prázdné.');
            }

            // --- Logika pro RANNÍ část ---
            if (fileRanni) {
                await uploadFile(fileRanni, 'ranni', initialData.files.ranni);
            } else if (initialData.files.ranni && (
                initialData.files.ranni.line_number !== formData.line_number ||
                initialData.files.ranni.course_number !== formData.course_number ||
                initialData.files.ranni.transport_type !== formData.transport_type ||
                initialData.files.ranni.day_type !== formData.day_type
            )) {
                // Pokud nebyl nahrán nový soubor, ale změnila se metadata, aktualizujeme DB záznam
                 const { error: updateError } = await supabase
                    .from('jizdni_rady')
                    .update({
                        line_number: formData.line_number,
                        course_number: formData.course_number,
                        transport_type: formData.transport_type,
                        day_type: formData.day_type,
                    })
                    .eq('id', initialData.files.ranni.id);

                if (updateError) throw new Error(`Chyba při aktualizaci RANNÍ DB metadat: ${updateError.message}`);
            }

            // --- Logika pro ODPOLEDNÍ část ---
            if (fileOdpoledni) {
                await uploadFile(fileOdpoledni, 'odpoledni', initialData.files.odpoledni);
            } else if (initialData.files.odpoledni && (
                initialData.files.odpoledni.line_number !== formData.line_number ||
                initialData.files.odpoledni.course_number !== formData.course_number ||
                initialData.files.odpoledni.transport_type !== formData.transport_type ||
                initialData.files.odpoledni.day_type !== formData.day_type
            )) {
                // Pokud nebyl nahrán nový soubor, ale změnila se metadata, aktualizujeme DB záznam
                 const { error: updateError } = await supabase
                    .from('jizdni_rady')
                    .update({
                        line_number: formData.line_number,
                        course_number: formData.course_number,
                        transport_type: formData.transport_type,
                        day_type: formData.day_type,
                    })
                    .eq('id', initialData.files.odpoledni.id);

                if (updateError) throw new Error(`Chyba při aktualizaci ODPOLEDNÍ DB metadat: ${updateError.message}`);
            }


            setStatusMessage('Změny byly úspěšně uloženy a soubory nahrány!');
            onSaveSuccess(); // Aktualizace seznamu na pozadí
            setTimeout(onClose, 2000); // Zavřít modal po úspěchu

        } catch (error: any) {
            console.error('Chyba při ukládání:', error);
            setStatusMessage(`CHYBA: ${error.message}`);
        } finally {
            setIsSaving(false);
        }
    };

    // Vykreslení
    return (
        <div
            className="fixed inset-0 z-50 flex items-center justify-center bg-black bg-opacity-70 backdrop-blur-sm p-4"
            onClick={onClose}
        >
            <div
                className="bg-white dark:bg-zinc-800 p-6 sm:p-8 rounded-xl shadow-2xl max-w-xl w-full max-h-[90vh] overflow-y-auto transform transition-all duration-300"
                onClick={(e) => e.stopPropagation()}
            >
                <h2 className="text-2xl font-bold mb-6 text-blue-600 dark:text-blue-300">
                    Editovat Jízdní Řád: {initialData.line_number}/{initialData.course_number}
                </h2>

                {/* FORMULÁŘ */}
                <div className="space-y-6">

                    {/* Skupina Metadat */}
                    <div className="grid grid-cols-2 gap-4">
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Číslo Kurzu:</label>
                            <input
                                type="text"
                                value={formData.line_number}
                                onChange={(e) => setFormData(prev => ({ ...prev, line_number: e.target.value.trim() }))}
                                className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"
                                disabled={isSaving}
                            />
                        </div>
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Číslo Linky:</label>
                            <input
                                type="text"
                                value={formData.course_number}
                                onChange={(e) => setFormData(prev => ({ ...prev, course_number: e.target.value.trim() }))}
                                className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"
                                disabled={isSaving}
                            />
                        </div>
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Typ Dopravy:</label>
                            <select
                                value={formData.transport_type}
                                onChange={(e) => setFormData(prev => ({ ...prev, transport_type: e.target.value as 'autobus' | 'tramvaj' }))}
                                className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"
                                disabled={isSaving}
                            >
                                <option value="autobus">Autobus 🚌</option>
                                <option value="tramvaj">Tramvaj 🚋</option>
                            </select>
                        </div>
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300">Dny Provozu:</label>
                            <select
                                value={formData.day_type}
                                onChange={(e) => setFormData(prev => ({ ...prev, day_type: e.target.value as 'pracovni' | 'vikend' }))}
                                className="mt-1 block w-full p-2 border border-gray-300 rounded-md dark:bg-zinc-700 dark:border-zinc-600 dark:text-zinc-100"
                                disabled={isSaving}
                            >
                                <option value="pracovni">Pracovní dny</option>
                                <option value="vikend">Víkend</option>
                            </select>
                        </div>
                    </div>

                    {/* Nahrávání souborů */}
                    <p className="text-lg font-semibold text-gray-800 dark:text-zinc-50 border-t pt-4 mt-4">Nahrát nové soubory (volitelné)</p>
                    <div className="grid grid-cols-2 gap-4">

                        {/* Ranní soubor */}
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300 flex items-center mb-1">
                                <Upload size={16} className="mr-1" /> Ranní JŘ ({initialData.files.ranni ? 'Nahrazeno' : 'Vytvořeno'}):
                            </label>
                            <input
                                type="file"
                                accept="image/png, image/jpeg, image/webp"
                                onChange={(e) => setFileRanni(e.target.files ? e.target.files[0] : null)}
                                className="mt-1 block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-3 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100 dark:file:bg-blue-800 dark:file:text-blue-100"
                                disabled={isSaving}
                            />
                            {initialData.files.ranni && !fileRanni && (
                                <p className="text-xs text-green-600 dark:text-green-400 mt-1">Stávající soubor bude použit, pokud nenahrajete nový.</p>
                            )}
                            {fileRanni && (
                                <p className="text-xs text-orange-600 dark:text-orange-400 mt-1">Nový soubor: **{fileRanni.name}** připraven k nahrání.</p>
                            )}
                        </div>

                        {/* Odpolední soubor */}
                        <div>
                            <label className="block text-sm font-medium text-gray-700 dark:text-zinc-300 flex items-center mb-1">
                                <Upload size={16} className="mr-1" /> Odpolední JŘ ({initialData.files.odpoledni ? 'Nahrazeno' : 'Vytvořeno'}):
                            </label>
                            <input
                                type="file"
                                accept="image/png, image/jpeg, image/webp"
                                onChange={(e) => setFileOdpoledni(e.target.files ? e.target.files[0] : null)}
                                className="mt-1 block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-3 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100 dark:file:bg-blue-800 dark:file:text-blue-100"
                                disabled={isSaving}
                            />
                            {initialData.files.odpoledni && !fileOdpoledni && (
                                <p className="text-xs text-green-600 dark:text-green-400 mt-1">Stávající soubor bude použit, pokud nenahrajete nový.</p>
                            )}
                            {fileOdpoledni && (
                                <p className="text-xs text-orange-600 dark:text-orange-400 mt-1">Nový soubor: **{fileOdpoledni.name}** připraven k nahrání.</p>
                            )}
                        </div>
                    </div>
                </div>

                {/* Zpráva o stavu */}
                {statusMessage && (
                    <div className={`mt-4 p-3 rounded-lg text-sm font-medium ${statusMessage.startsWith('CHYBA') ? 'bg-red-100 text-red-700 dark:bg-red-900 dark:text-red-300' : 'bg-green-100 text-green-700 dark:bg-green-900 dark:text-green-300'}`}>
                        {statusMessage}
                    </div>
                )}

                {/* Tlačítka */}
                <div className="mt-6 flex justify-end space-x-3">
                    <button
                        onClick={onClose}
                        className="px-4 py-2 bg-gray-200 dark:bg-zinc-700 text-gray-800 dark:text-zinc-100 rounded-lg hover:bg-gray-300 dark:hover:bg-zinc-600 transition disabled:opacity-50"
                        disabled={isSaving}
                    >
                        Zavřít
                    </button>
                    <button
                        onClick={handleSave}
                        className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition disabled:opacity-50 flex items-center"
                        disabled={isSaving || (!fileRanni && !fileOdpoledni &&
                            formData.line_number === initialData.line_number &&
                            formData.course_number === initialData.course_number &&
                            formData.transport_type === initialData.transport_type &&
                            formData.day_type === initialData.day_type
                        )}
                    >
                        {isSaving ? (
                            <>
                                <Loader2 size={18} className="animate-spin mr-2" /> Ukládám...
                            </>
                        ) : 'Uložit Změny'}
                    </button>
                </div>
            </div>
        </div>
    );

};

const { user, isLoading: isAuthLoading, status, role } = useAuth();
const isAdmin = role === 'admin';

    // ... (ostatní stavy a filtry zůstávají stejné) ...
    const [jizdniRady, setJizdniRady] = useState<GroupedTimetable[]>([]);
    const [isLoadingData, setIsLoadingData] = useState(true);
    const [error, setError] = useState<string | null>(null);
    const [isModalOpen, setIsModalOpen] = useState(false);
    const [modalImageUrl, setModalImageUrl] = useState<string | null>(null);
    const [modalAltText, setModalAltText] = useState('');
    const [isEditModalOpen, setIsEditModalOpen] = useState(false);
    const [editData, setEditData] = useState<GroupedTimetable | null>(null);

    const initialFilters: Filters ={
        transport_type: 'all',
        day_type: 'all',
        line_number: '',
        course_number: '',
    };

    const [draftFilters, setDraftFilters] = useState<Filters>(initialFilters);
    const [appliedFilters, setAppliedFilters] = useState<Filters>(initialFilters);

    // Funkce pro fetchování dat je klíčová a musí se volat po editaci/mazání
    const fetchJizdniRady = useCallback(async () => {
      setIsLoadingData(true);
      setError(null);

      let query = supabase
        .from('jizdni_rady')
        .select('*')
        .order('line_number', { ascending: true })
        .order('course_number', { ascending: true });

      if (appliedFilters.transport_type !== 'all') {
        query = query.eq('transport_type', appliedFilters.transport_type);
      }
      if (appliedFilters.day_type !== 'all') {
        query = query.eq('day_type', appliedFilters.day_type);
      }
      if (appliedFilters.line_number) {
        query = query.ilike('line_number', `%${appliedFilters.line_number}%`);
      }
      if (appliedFilters.course_number) {
        query = query.ilike('course_number', `%${appliedFilters.course_number}%`);
      }

      const { data, error: fetchError } = await query;
        // ... (Logika seskupování dat zůstává stejná) ...
        if (fetchError) {
            setError('Nepodařilo se načíst jízdní řády. Zkontrolujte RLS politiky.');
            setJizdniRady([]);
        } else {
            const groupedData = (data as JizdniRad[]).reduce((acc, current) => {
                const key = `${current.line_number}-${current.course_number}-${current.day_type}`;
                if (!acc[key]) {
                    acc[key] = {
                        line_number: current.line_number,
                        course_number: current.course_number,
                        transport_type: current.transport_type,
                        day_type: current.day_type,
                        created_at: current.created_at,
                        files: { ranni: null, odpoledni: null },
                    };
                }
                if (current.part_of_day === 'ranni') {
                    acc[key].files.ranni = current;
                } else if (current.part_of_day === 'odpoledni') {
                    acc[key].files.odpoledni = current;
                }
                return acc;
            }, {} as Record<string, GroupedTimetable>);

            const finalDisplayData = Object.values(groupedData).sort((a, b) =>
                a.line_number.localeCompare(b.line_number) || a.course_number.localeCompare(b.course_number)
            );
            setJizdniRady(finalDisplayData);
        }
      setIsLoadingData(false);
    }, [appliedFilters]);

    useEffect(() => {
      if (!isAuthLoading && user) {
          fetchJizdniRady();
      }
    }, [isAuthLoading, user, fetchJizdniRady]);

    // Ostatní callbacky (handleFilterChange, handleSearch, openModal, closeModal) zůstávají stejné

    const openEditModal = useCallback((data: GroupedTimetable) => {
        setEditData(data);
        setIsEditModalOpen(true);
    }, []);

    const closeEditModal = useCallback(() => {
        setIsEditModalOpen(false);
        setEditData(null);
    }, []);

    const openModal = useCallback((url: string, alt: string) => {
        setModalImageUrl(url);
        setModalAltText(alt);
        setIsModalOpen(true);
    }, []);

    const closeModal = useCallback(() => {
        setIsModalOpen(false);
        setModalImageUrl(null);
        setModalAltText('');
    }, []);

    // Funkce pro MAZÁNÍ (z předchozí odpovědi)
    const handleDelete = useCallback(async (jizdniRad: GroupedTimetable) => {
        if (!window.confirm(`Opravdu chcete smazat jízdní řád ${jizdniRad.line_number}/${jizdniRad.course_number} (${jizdniRad.day_type})? Tato akce smaže POUZE záznamy v databázi a soubory v Supabase Storage.`)) {
            return;
        }

        setIsLoadingData(true);
        setError(null);

        try {
            const filesToDelete: { id: string, image_url: string }[] = [];

            if (jizdniRad.files.ranni) {
                filesToDelete.push({ id: jizdniRad.files.ranni.id, image_url: jizdniRad.files.ranni.image_url });
            }
            if (jizdniRad.files.odpoledni) {
                filesToDelete.push({ id: jizdniRad.files.odpoledni.id, image_url: jizdniRad.files.odpoledni.image_url });
            }

            // 1. Smazání souborů ze Supabase Storage
             for (const file of filesToDelete) {
                const urlParts = file.image_url.split('/storage/v1/object/public/jizdni-rady/');
                if (urlParts.length === 2) {
                    const filePath = urlParts[1];
                    const { error: storageError } = await supabase.storage
                        .from('jizdni-rady')
                        .remove([filePath]);
                    if (storageError) console.error(`Chyba při mazání souboru ${filePath}:`, storageError.message);
                }
            }

            // 2. Smazání z databáze (jizdni_rady)
            const idsToDelete = filesToDelete.map(f => f.id);
            if (idsToDelete.length > 0) {
                const { error: dbError } = await supabase
                    .from('jizdni_rady')
                    .delete()
                    .in('id', idsToDelete);

                if (dbError) throw new Error(`Chyba při mazání záznamů v DB: ${dbError.message}`);
            }

            // 3. Aktualizace UI
            alert('Jízdní řád byl úspěšně smazán!');
            await fetchJizdniRady();

        } catch (e: any) {
            setError(`Chyba při mazání: ${e.message}`);
            setIsLoadingData(false);
        }

    }, [fetchJizdniRady]);


    // Ochrana přístupu (zůstává stejná)
    if (isAuthLoading || isLoadingData) { /* ... Loading UI ... */
        return (
            <div className="flex justify-center items-center min-h-screen bg-gray-50 dark:bg-gray-900">
            <GridLoader color="#0066ff" size={15} />
            </div>
        );
    }
    if (error) { /* ... Error UI ... */
        return (
            <div className="p-4 bg-red-100 dark:bg-red-900 text-red-700 dark:text-red-300 rounded mx-auto max-w-4xl mt-8">
                <h1 className="font-bold">Chyba:</h1> {error}
            </div>
        );
    }
    if (!user || status !== 'approved') { /* ... Access Denied UI ... */
        return (
            <div className="p-8 text-center bg-white dark:bg-zinc-800 rounded-lg shadow-xl mx-auto max-w-md mt-10">
                <h1 className="text-2xl text-red-600 dark:text-red-400 font-bold">Přístup odepřen</h1>
                <p className="text-gray-600 dark:text-zinc-400 mt-2">Musíte být schválený uživatel pro přístup k jízdním řádům.</p>
            </div>
        );
    }


    return (
      <div className="min-h-screen bg-gray-100 dark:bg-gray-900 p-4 sm:p-8">
        <div className="max-w-7xl mx-auto">
          <h1 className="text-2xl sm:text-4xl font-bold mb-6 sm:mb-8 text-gray-800 dark:text-zinc-50 border-b-2 border-blue-700 pb-2 flex items-center">
            <span className='text-2xl'>Jízdní Řády</span> {isAdmin && <span className="text-2xl text-red-500">(ADMIN)</span>}
          </h1>

          <JizdniRadFilters
            filters={draftFilters}
            onFilterChange={(newFilters: Partial<Filters>) => setDraftFilters(prev => ({ ...prev, ...newFilters }))}
            onSearch={() => setAppliedFilters(draftFilters)}
            isLoading={isLoadingData}
          />

          {jizdniRady.length === 0 ? (
            <div className="bg-white dark:bg-zinc-800 p-6 rounded-xl shadow-lg text-center text-gray-600 dark:text-zinc-400">
              Nebyly nalezeny žádné jízdní řády odpovídající vašim kritériím.

                  {isAdmin && (

                              <Edit size={16} />

'use client';

import React, { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/lib/supabase/client';
import { useAuth } from '@/context/AuthContext';
import { X, Bell, Loader2 } from 'lucide-react';

// --- TYPY DAT ---
interface Announcement {
id: string;
title: string;
content: string;
priority: number;
}

const AnnouncementBanner: React.FC = () => {
const { user, status, role, isLoading: isAuthLoading } = useAuth();
const [unreadAnnouncements, setUnreadAnnouncements] = useState<Announcement[]>([]);
const [isLoading, setIsLoading] = useState(true);
const [isDismissing, setIsDismissing] = useState<string | null>(null); // ID právě odstraňovaného oznámení

    // Funkce pro načtení nepřečtených oznámení
    const fetchUnreadAnnouncements = useCallback(async () => {
        if (isAuthLoading || !user || status !== 'approved') {
            setIsLoading(false);
            setUnreadAnnouncements([]);
            return;
        }

        setIsLoading(true);

        // 1. Určení cílové role pro dotaz
        const targetRoles = ['all', role].filter((r): r is 'all' | 'admin' | 'user' => r === 'all' || r === 'admin' || r === 'user');

        // 2. Načtení ID oznámení, které uživatel JIZ PŘEČETL
        const { data: readIdsData, error: readError } = await supabase
            .from('user_announcement_reads')
            .select('announcement_id')
            .eq('user_id', user.id); // Klíčové: filtrujeme na aktuálního uživatele

        if (readError) {
            console.error('Chyba při načítání přečtených ID:', readError);
            setIsLoading(false);
            return;
        }

        const readAnnouncementIds = readIdsData?.map(r => r.announcement_id) || [];

        // Zde použijeme "NOT IN" na straně Supabase, pokud je seznam čtených ID neprázdný
        let query = supabase
            .from('portal_announcements')
            .select(`id, title, content, priority`)
            .eq('is_active', true)
            .in('target_role', targetRoles)
            .order('priority', { ascending: false });

        // 3. Aplikace filtru pro nepřečtené (pomocí .not().in())
        if (readAnnouncementIds.length > 0) {
            query = query.not('id', 'in', `(${readAnnouncementIds.join(',')})`);
        }

        const { data: announcements, error: fetchError } = await query;

        if (fetchError) {
            console.error('Chyba při načítání oznámení:', fetchError);
            setIsLoading(false);
            return;
        }

        // 4. Nastavení nepřečtených oznámení
        setUnreadAnnouncements(announcements as Announcement[] || []);
        setIsLoading(false);

    }, [isAuthLoading, user, status, role]);

    useEffect(() => {
        fetchUnreadAnnouncements();
    }, [fetchUnreadAnnouncements]);

    // Funkce pro označení oznámení jako přečteného (Dismiss)
    const handleDismiss = useCallback(async (announcementId: string) => {
        if (!user) return;

        setIsDismissing(announcementId);

        try {
            // Vložení záznamu do propojovací tabulky
            const { error: insertError } = await supabase
                .from('user_announcement_reads')
                .insert({
                    user_id: user.id, // Klíčové: Uložení vazby na AKTUALNIHO UŽIVATELE
                    announcement_id: announcementId,
                });

            if (insertError) throw insertError;

            // Odebrání z lokálního stavu
            setUnreadAnnouncements(prev => prev.filter(a => a.id !== announcementId));

        } catch (err) {
            console.error('Chyba při označování jako přečteno:', err);
            // Můžete přidat zobrazení chyby uživateli
        } finally {
            setIsDismissing(null);
        }
    }, [user]);

    // Ochrana: Nejsou oznámení, čekáme na ověření, nebo je uživatel blokovaný
    if (isAuthLoading || isLoading || unreadAnnouncements.length === 0 || status !== 'approved') {
        return null;
    }

    // Zobrazení všech nepřečtených oznámení (seřazených podle priority)
    return (
        <div className="w-full max-w-7xl mx-auto p-4 space-y-4">
            {unreadAnnouncements.map((announcement) => (
                <div
                    key={announcement.id}
                    className="flex items-start bg-blue-50 dark:bg-blue-900 border-l-4 border-blue-600 dark:border-blue-400 p-4 rounded-lg shadow-md transition-all duration-300"
                >
                    <Bell size={20} className="text-blue-600 dark:text-blue-400 mt-0.5 mr-3 flex-shrink-0" />

                    <div className="flex-grow">
                        <h4 className="text-lg font-bold text-blue-800 dark:text-blue-200">
                           {announcement.title}
                           {announcement.priority > 1 && <span className="ml-2 text-xs font-semibold bg-red-500 text-white px-2 py-0.5 rounded-full">DŮLEŽITÉ</span>}
                        </h4>
                        <p className="text-sm text-blue-700 dark:text-blue-300 mt-1 whitespace-pre-wrap">
                            {announcement.content}
                        </p>
                    </div>

                    <button
                        onClick={() => handleDismiss(announcement.id)}
                        disabled={isDismissing === announcement.id}
                        className="ml-4 flex-shrink-0 p-1 rounded-full text-blue-600 hover:bg-blue-200 dark:text-blue-400 dark:hover:bg-blue-800 transition disabled:opacity-50"
                        title="Označit jako přečteno"
                    >
                        {isDismissing === announcement.id ? <Loader2 size={20} className="animate-spin" /> : <X size={20} />}
                    </button>
                </div>
            ))}
        </div>
    );

};

export default AnnouncementBanner;

const fetchUserProfileData = async (user: User | null): Promise<ProfileData> => {
// 1. Nepřihlášený uživatel
if (!user) {
return { role: 'ghost', full_name: null, status: 'pending', is_blocked: false };
}

// 2. Získání role A jména z tabulky 'profiles'
const { data, error } = await supabase
.from('profiles')
// POZOR ZDE: Změna selectu, aby stahoval obě hodnoty
.select('role, full_name, status, is_blocked, email')
.eq('id', user.id)
.single();
//console.log('Fetched profile data:', data, 'Error:', error);
if (error || !data) {
console.error('Chyba při získávání dat profilu z databáze:', error?.message);
// Bezpečný fallback: defaultní role a prázdné jméno
return { role: 'user' , full_name: user.email?.split('@')[0] ?? null, status: 'pending', is_blocked: false };
}

// 3. Kontrola a vrácení dat
const userRole = data.role as Role;
const fullName = data.full_name;
const status = data.status;
const isBlocked = data.is_blocked;

return {
// Zajištění, že role je vždy 'user' nebo 'admin' (nebo fallback)
role: (userRole === 'admin') ? 'admin' : 'user',
full_name: fullName,
status,
is_blocked: isBlocked,
};
};

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
const [session, setSession] = useState<Session | null>(null);
const [user, setUser] = useState<User | null>(null);
const [full_name, setFullName] = useState<string | null>(null);
const [role, setRole] = useState<Role>('ghost');
const [isLoading, setIsLoading] = useState(true);
const [status, setStatus] = useState<'pending' | 'approved'>('pending');
const [is_blocked, setIsBlocked] = useState<boolean>(false);

// Funkce, která provede celou logiku načítání a aktualizace stavu
const handleAuthStateChange = useCallback(async (currentSession: Session | null) => {
const currentUser = currentSession?.user ?? null;

    setSession(currentSession);
    setUser(currentUser);

    // Asynchronní volání pro získání role A jména
    const profileData = await fetchUserProfileData(currentUser);

    setRole(profileData.role);
    setFullName(profileData.full_name); // <-- AKTUALIZACE JMENA
    setStatus(profileData.status);
    setIsBlocked(profileData.is_blocked);
    setIsLoading(false);

}, []);

useEffect(() => {
// Inicializace a posluchač (zůstává stejné)
supabase.auth.getSession().then(({ data: { session } }) => {
handleAuthStateChange(session);
});

    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      (event, session) => {
        console.log('Auth State Changed:', event);
        handleAuthStateChange(session);
      }
    );

    return () => {
      subscription.unsubscribe();
    };

}, [handleAuthStateChange]);
//console.log('AuthContext State:', { session, user, role, full_name, status, is_blocked, isLoading });
const value: AuthState = {
session,
user,
full_name, // <-- EXPORTOVÁNÍ JMENA
role,
status,
is_blocked,
isLoading,
};

// Dokud se nenačte počáteční stav (včetně role a jména), zobrazíme loading
if (isLoading) {
return <div className="flex items-center justify-center h-screen bg-gray-50"><p className="text-xl text-gray-500">Načítání uživatelského profilu...</p></div>;
}

return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
};

// Custom Hook (zůstává stejný)
export const useAuth = () => {
const context = useContext(AuthContext);
if (context === undefined) {
throw new Error('useAuth musí být použit uvnitř AuthProvider');
}
return context;
};
